class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

n1 = Node(10)
n2 = Node(20)
n3 = Node(30)
n1.next = n2
n2.next = n3
head = n1

# Traversal
current = head
while current is not None:
    print(current.data)
    current = current.next

Key idea: move current, not head. This preserves the starting reference to the list.
